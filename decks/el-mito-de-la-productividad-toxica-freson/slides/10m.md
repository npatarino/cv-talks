      <script>
      (function () {
        var cnt = document.getElementById('cnt10');
        var cells = document.querySelectorAll('#days12 .days-bar i');
        if (!cnt) return;
        var T = 2600, t0 = null;
        var ease = function (x) { return 1 - Math.pow(1 - x, 3); };
        function frame(t) {
          if (t0 === null) t0 = t;
          var k = Math.min(1, (t - t0) / T), e = ease(k);
          cnt.textContent = Math.round(10 * e);
          var on = Math.round(12 * e);
          cells.forEach(function (c, i) { c.classList.toggle('on', i < on); });
          if (k < 1) requestAnimationFrame(frame);
        }
        cnt.textContent = '0';
        requestAnimationFrame(frame);
      })();
      </script>
